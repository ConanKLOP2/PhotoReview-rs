## Public API
Compiling harness shape for later implementation; illustrative only, not added as repo file:
```rust
use std::path::PathBuf;

pub fn corpus_dir() -> Option<PathBuf> {
    if let Some(path) = std::env::var_os("PHOTOREVIEW_RS_RAW_CORPUS_DIR") {
        return Some(PathBuf::from(path));
    }
    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
    Some(root.parent()?.parent()?.join("tests").join("fixtures").join("raw-corpus"))
}

pub fn strict() -> bool {
    matches!(std::env::var("PHOTOREVIEW_RS_RAW_CORPUS_STRICT").as_deref(),
        Ok("1") | Ok("true") | Ok("TRUE"))
}

pub fn require_sample(name: &str) -> Option<PathBuf> {
    let path = corpus_dir()?.join(name);
    if path.is_file() { Some(path) }
    else if strict() { panic!("required RAW corpus sample missing: {name}") }
    else { None }
}
```
